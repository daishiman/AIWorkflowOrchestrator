 *
 * 各AIプロバイダーの公式/ブランドアイコンを表示

  provider,
  size = 18,
  className,

  const iconClass = clsx("flex-shrink-0", className);

  switch (provider) {

    case "anthropic":

    default:
      return null;
  }
};
