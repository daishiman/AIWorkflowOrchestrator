**実装パターン**:

```typescript
// パストラバーサル検出
if (dirPath.includes("..")) {
  return { valid: false, error: "directory traversal not allowed" };
}

// URLエンコードされたトラバーサル検出
if (dirPath.includes("%2e%2e") || dirPath.includes("%2E%2E")) {
  return { valid: false, error: "encoded traversal not allowed" };
}
```
