## 設計検証チェックリスト

### Zustand参照安定性の検証

```typescript
// 検証コード例
const useStore = create<TestState>((set) => ({
  count: 0,
  increment: () => set((s) => ({ count: s.count + 1 })),
}));

// 個別セレクタ
const useIncrement = () => useStore((s) => s.increment);

// テストコンポーネント
const TestComponent = () => {
  const increment = useIncrement();
  const ref = useRef(increment);

  useEffect(() => {
    // 参照が安定していることを検証
    console.assert(ref.current === increment, "参照が変わっている");
    ref.current = increment;
  }, [increment]);

  // increment を依存配列に含めても無限ループしないことを検証
  useEffect(() => {
    console.log("This should run only once");
  }, [increment]);

  return null;
};
```
