## 7. 検証スクリプト

```bash
#!/bin/bash
# acceptance-check.sh - 受け入れ基準自動検証スクリプト

echo "=== アーキテクチャ準拠チェック ==="

# Domain層のインフラ依存チェック
echo "Checking Domain layer dependencies..."
if grep -r "drizzle" packages/shared/src/features/chat-history/domain/ 2>/dev/null; then
  echo "❌ ARCH-D02: Domain層にDrizzle依存が見つかりました"
  exit 1
else
  echo "✅ ARCH-D02: Domain層はDrizzle非依存"
fi

echo ""
echo "=== コード品質チェック ==="

# TypeScript型チェック
echo "Running TypeScript check..."
if pnpm --filter @repo/shared typecheck; then
  echo "✅ QA-S01: TypeScript型エラー0件"
else
  echo "❌ QA-S01: TypeScript型エラーあり"
  exit 1
fi

# ESLintチェック
echo "Running ESLint..."
if pnpm --filter @repo/shared lint; then
  echo "✅ QA-S02: ESLintエラー0件"
else
  echo "❌ QA-S02: ESLintエラーあり"
  exit 1
fi

echo ""
echo "=== テスト実行 ==="

# ユニットテスト
echo "Running unit tests..."
if pnpm --filter @repo/shared test; then
  echo "✅ QA-T01: 全ユニットテストPASS"
else
  echo "❌ QA-T01: テスト失敗"
  exit 1
fi

echo ""
echo "=== 全チェック完了 ==="
echo "受け入れ基準を満たしています"
```
