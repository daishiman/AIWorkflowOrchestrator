### 5.2 SQLクエリ例

```sql
-- 1. 既存エンティティ検索
SELECT * FROM entities
WHERE normalized_name = ?
  AND deleted_at IS NULL;

-- 2a. 既存エンティティ更新
UPDATE entities
SET importance_score = MAX(importance_score, ?),
    mention_count = mention_count + 1,
    updated_at = ?
WHERE id = ?;

-- 2b. 新規エンティティ挿入
INSERT INTO entities (
  id, name, type, description, embedding,
  importance_score, mention_count,
  first_seen_at, last_seen_at, metadata,
  created_at, updated_at
) VALUES (?, ?, ?, ?, ?, ?, 1, ?, ?, ?, ?, ?);

-- 3. チャンク-エンティティ関連付け
INSERT INTO chunk_entities (chunk_id, entity_id, mention_count, positions)
VALUES (?, ?, ?, ?)
ON CONFLICT (chunk_id, entity_id)
DO UPDATE SET
  mention_count = chunk_entities.mention_count + excluded.mention_count,
  positions = excluded.positions;
```
